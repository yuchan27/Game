extends Node

# Ordinary isolated art states only: no interactions, resets, rewards or saves.
func _ready() -> void:
    call_deferred("_capture")

func _capture() -> void:
    if not ProjectSettings.get_setting("application/config/use_custom_user_dir", false):
        push_error("Use an isolated art project with dedicated custom user data.")
        get_tree().quit(2)
        return
    var output := ""
    var selected := PackedStringArray()
    for argument in OS.get_cmdline_user_args():
        if argument.begins_with("--art-output="):
            output = argument.trim_prefix("--art-output=")
        elif argument.begins_with("--art-view="):
            selected = argument.trim_prefix("--art-view=").split(",", false)
    if output.is_empty() or not output.is_absolute_path():
        push_error("Pass an absolute --art-output directory.")
        get_tree().quit(2)
        return
    DirAccess.make_dir_recursive_absolute(output)
    print("LATE_COURT_ART_VIEWPORT: ", get_viewport().get_visible_rect().size)
    var views = [
        ["factory-story-approach", "wasteland", Vector2(3727.36,2421.76), false, "old_factory"],
        ["factory-south-approach", "wasteland", Vector2(3727.36,2600), false, "old_factory"],
        ["factory-forecourt", "wasteland", Vector2(3900,2500), false, "old_factory"],
        ["factory-map-staged", "wasteland", Vector2(3727.36,2421.76), false, "old_factory"],
        ["city-story-approach", "wasteland", Vector2(3712,3072), false, "ruined_metropolis"],
        ["city-south-approach", "wasteland", Vector2(3712,3240), false, "ruined_metropolis"],
        ["city-archive-forecourt", "wasteland", Vector2(3890,3170), false, "ruined_metropolis"],
        ["city-map-staged", "wasteland", Vector2(3712,3072), false, "ruined_metropolis"],
        ["facility-story-approach", "wasteland", Vector2(3328,2826.24), false, "underground_facility"],
        ["facility-south-approach", "wasteland", Vector2(3328,3000), false, "underground_facility"],
        ["facility-vault-forecourt", "wasteland", Vector2(3500,2920), false, "underground_facility"],
        ["facility-map-staged", "wasteland", Vector2(3328,2826.24), false, "underground_facility"],
        ["convoy-control", "wasteland", Vector2(3584,3225.6)],
        ["marsh-control", "wasteland", Vector2(3061.76,3002.88), false, "toxic_marsh"],
        ["crystal-control", "wasteland", Vector2(3456,3650.56), false, "crystal_scar"]
    ]
    for view: Array in views:
        if not selected.is_empty() and not view[0] in selected:
            continue
        get_tree().paused = false
        GameState.active_quest_id = ""
        GameState.quest_progress = {}
        GameState.completed_quests.clear()
        if view.size() > 3 and view[3]:
            GameState.completed_quests.append("facility_memory")
        GameState.active_spawn_point = "default"
        GameState.current_route_id = "scrap_highway"
        if view.size() > 4:
            GameState.current_route_id = view[4]
        var scene = load("res://scenes/levels/%s/%s.tscn" % [view[1], String(view[1]).capitalize()]).instantiate()
        get_tree().root.add_child(scene)
        get_tree().current_scene = scene
        var actor = get_tree().get_first_node_in_group("map_player")
        if actor == null:
            push_error("Regional art scene did not create its player: " + String(view[0]))
            get_tree().quit(2)
            return
        actor.global_position = view[2]
        actor.reset_physics_interpolation()
        var camera = actor.get_node("Camera2D")
        camera.position_smoothing_enabled = false
        camera.reset_smoothing()
        camera.force_update_scroll()
        # Allow broadphase overlap notifications to settle before freezing the art state.
        for frame in range(16):
            await get_tree().physics_frame
        await get_tree().process_frame
        if not String(view[0]).ends_with("centre"):
            var notice = scene.find_child("NoticeLabel", true, false)
            if notice != null:
                notice.hide()
        if String(view[0]).ends_with("map-staged"):
            var hud = get_tree().get_first_node_in_group("hud")
            hud.set_fullscreen_map(true)
            await get_tree().process_frame
            await get_tree().process_frame
        get_tree().paused = true
        await get_tree().process_frame
        await RenderingServer.frame_post_draw
        get_viewport().get_texture().get_image().save_png(output.path_join(String(view[0]) + ".png"))
        print("LATE_COURT_ART_FRAME: ", view[0], " camera=", camera.get_screen_center_position(), " zoom=", camera.zoom, " actor=", actor.global_position)
        if view[1] == "wasteland":
            print("LATE_COURT_ROUTE: ", scene.route_id)
        for child in scene.get_children():
            if child is WorldBackdrop:
                var ground_material: ShaderMaterial = child.get("_ground_material")
                print("LATE_COURT_GROUND: water_enabled=", ground_material.get_shader_parameter("water_enabled"), " water_size=", ground_material.get_shader_parameter("water_size"), " earth_contrast=", ground_material.get_shader_parameter("earth_contrast"))
        var capsule = actor.get_node("CollisionShape2D")
        print("LATE_COURT_PLAYER_FOOT: offset=", capsule.position, " radius=", capsule.shape.radius, " height=", capsule.shape.height)
        for station in get_tree().get_nodes_in_group("map_station"):
            var source_id = String(station.get_meta("structure_asset_id", ""))
            for child in station.get_children():
                if child is Sprite2D:
                    print("LATE_COURT_SOURCE: ", source_id, " atlas=", child.texture is AtlasTexture,
                        " crop=", child.texture.region if child.texture is AtlasTexture else Rect2(),
                        " contact=", child.material.get_shader_parameter("ground_contact_enabled") if child.material is ShaderMaterial else false)
        for prop in get_tree().get_nodes_in_group("world_prop"):
            if prop.prop_id in ["highway_bus", "marsh_pump", "crystal_spire", "factory_turbine", "evacuation_archive", "memory_vault"]:
                print("LATE_COURT_LANDMARK: id=", prop.prop_id, " position=", prop.global_position, " collision=", prop.get_collision_bounds(), " visual=", prop.get_visual_bounds())
                for child in prop.get_children():
                    if child is Sprite2D:
                        print("LATE_COURT_LANDMARK_SPRITE: origin=", child.position, " scale=", child.scale, " texture_size=", child.texture.get_size(), " fade_alpha=", child.self_modulate.a)
        for body in get_tree().get_nodes_in_group("world_architecture"):
            if body.global_position.distance_to(actor.global_position) < 700.0:
                print("LATE_COURT_ARCHITECTURE: ", body.name, " position=", body.global_position)
        var capture_hud = get_tree().get_first_node_in_group("hud")
        var hint = capture_hud.controls_hint
        print("LATE_COURT_CONTEXT: drawn=", hint.is_visible_in_tree(), " text=", hint.text,
            " screen=", hint.get_global_transform_with_canvas() * Vector2.ZERO, " size=", hint.size)
        scene.queue_free()
        await get_tree().process_frame
    get_tree().quit()
