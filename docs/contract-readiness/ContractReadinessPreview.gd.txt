extends Node

# Ordinary staged UI composition; no hand-in, action completion, IO or QA.
func _ready() -> void:
	process_mode = Node.PROCESS_MODE_ALWAYS
	call_deferred("_capture")

func _capture() -> void:
	if not ProjectSettings.get_setting("application/config/use_custom_user_dir", false) or ProjectSettings.get_setting("application/config/custom_user_dir_name", "") != "R17-Ordinary-Art-Preview-20261023":
		push_error("Use the isolated presentation project.")
		get_tree().quit(2)
		return
	var output := ""
	for argument in OS.get_cmdline_user_args():
		if argument.begins_with("--contract-output="):
			output = argument.trim_prefix("--contract-output=")
	if output.is_empty() or not output.is_absolute_path():
		push_error("Pass an absolute --contract-output directory.")
		get_tree().quit(2)
		return
	DirAccess.make_dir_recursive_absolute(output)
	for view: String in ["city-bank-short", "city-provisional", "city-atlas", "city-clue-pending", "factory-boss", "legacy-ready", "city-ready", "city-last-outing-short"]:
		get_tree().paused = true
		GameState.reset_new_run(false)
		_stage(view)
		var scene = load(SceneRouter.SCENES[GameState.current_scene_id]).instantiate()
		get_tree().root.add_child(scene)
		get_tree().current_scene = scene
		var actor = get_tree().get_first_node_in_group("map_player")
		if actor != null:
			var camera = actor.get_node("Camera2D")
			camera.position_smoothing_enabled = false
			camera.reset_smoothing()
			camera.force_update_scroll()
		var hud = get_tree().get_first_node_in_group("hud")
		if view == "city-atlas":
			hud.call("_set_atlas_visible", true)
		else:
			hud.call("_set_contract_visible", true)
		for frame in range(8):
			await get_tree().process_frame
		await RenderingServer.frame_post_draw
		var error := get_viewport().get_texture().get_image().save_png(output.path_join(view + ".png"))
		print("CONTRACT_PRESENTATION_COPY: ", view, " ready=", GameState.is_active_quest_ready(), " hint=", GameState.get_journey_hint(), " checklist=", GameState.active_quest_requirements_text().replace("\n", " | "))
		print("CONTRACT_PRESENTATION_FRAME: ", view, " viewport=", get_viewport().get_visible_rect().size, " image_result=", error)
		scene.queue_free()
		await get_tree().process_frame
	get_tree().quit(0)

func _stage(view: String) -> void:
	var quest_id := "old_factory_core" if view == "factory-boss" else ("clear_scrap_route" if view == "legacy-ready" else "metropolis_archive")
	var quest := DataRegistry.get_quest(quest_id)
	var in_field := view in ["city-provisional", "city-atlas", "city-last-outing-short"]
	GameState.current_scene_id = "wasteland" if in_field else "village"
	GameState.current_route_id = String(quest["route"])
	GameState.active_quest_id = quest_id
	GameState.quest_progress = {"defeat_enemy": 8 if view == "factory-boss" else (3 if in_field else 5), "defeat_boss": 0}
	GameState.inventory["scrap"] = 12
	GameState.inventory["mutant_core"] = 2
	GameState.inventory["precision_chip"] = 1 if in_field or view == "city-bank-short" else 2
	for previous: Dictionary in DataRegistry.quests:
		if bool(previous.get("main_campaign", false)) and int(previous.get("chapter", 0)) < int(quest.get("chapter", 0)):
			GameState.completed_quests.append(String(previous["id"]))
	if view == "legacy-ready":
		return
	var banked := {}
	var outing_index := 1 if in_field else 2
	for index in range(outing_index):
		banked.merge(_results(quest["campaign_outings"][index]), true)
	GameState.quest_progress["field_objective_version"] = 1
	GameState.quest_progress["campaign"] = {"version": 1, "outing_index": outing_index, "actions": banked}
	if quest_id == "metropolis_archive":
		var key := "ruined_metropolis:story_metropolis_archive"
		GameState.discovered_events.append(key)
		if in_field or view == "city-clue-pending":
			GameState._expedition.uncommitted_clues.append(key)
	# These main quests have no legacy field objective marker; mirror real accepts.
	if not quest.has("field_objective"):
		GameState.quest_progress.erase("field_objective_version")
	if in_field:
		GameState._expedition.begin(GameState.current_route_id, quest_id)
		GameState._expedition.active["campaign_outing"] = quest["campaign_outings"][1]["id"]
		GameState._expedition.active["campaign_actions"] = _results(quest["campaign_outings"][1])
		GameState._expedition.active["counters"] = {"defeat_enemy": 2}
		if view != "city-last-outing-short":
			GameState._expedition.active["haul"] = {"precision_chip": 1}

func _results(outing: Dictionary) -> Dictionary:
	var results := {}
	for definition: Dictionary in outing["actions"]:
		results[definition["id"]] = definition["options"][0]["id"] if definition.get("type", "") == "choice" else "done"
	return results
